"""
linalg.py - linear algebra from scratch, in plain Python.

No NumPy, no imports beyond the standard library. A matrix is a list of rows,
a vector is a list of numbers. Entries may be ints, floats or Fractions; use
frac(A) to switch a matrix to exact fractions so that 1/3 prints as 1/3.

Each routine follows the method taught in MIT 18.06 (Strang), lectures 1-7:

    matmul, matvec, transpose          lectures 1, 3, 5
    eliminate, back_substitute, solve  lecture 2
    inverse (Gauss-Jordan)             lecture 3
    lu, lu_solve                       lectures 4, 5  (PA = LU)
    rref, rank, nullspace              lectures 6, 7

Run any lecture script in this folder to see them used on a robot.
"""
from fractions import Fraction
import math

EPS = 1e-10


# ----------------------------------------------------------------- basics
def frac(A):
    """Copy of a matrix or vector with every entry as an exact Fraction."""
    if A and isinstance(A[0], (list, tuple)):
        return [[Fraction(x).limit_denominator(10**6) for x in row] for row in A]
    return [Fraction(x).limit_denominator(10**6) for x in A]


def copy(A):
    return [list(row) for row in A]


def zeros(m, n):
    return [[0] * n for _ in range(m)]


def identity(n):
    return [[1 if i == j else 0 for j in range(n)] for i in range(n)]


def shape(A):
    return len(A), len(A[0])


def transpose(A):
    return [[A[i][j] for i in range(len(A))] for j in range(len(A[0]))]


def column(A, j):
    return [row[j] for row in A]


def from_columns(cols):
    return transpose(cols)


def matvec(A, x):
    """A times x. Each entry is a row of A dotted with x (the row way)."""
    return [sum(a * xi for a, xi in zip(row, x)) for row in A]


def combine(cols, weights):
    """Linear combination of column vectors (the column way of A times x)."""
    out = [0] * len(cols[0])
    for col, w in zip(cols, weights):
        out = [o + w * c for o, c in zip(out, col)]
    return out


def matmul(A, B):
    """C = AB, entry by entry: c_ij = (row i of A) . (column j of B)."""
    m, p, n = len(A), len(B), len(B[0])
    assert len(A[0]) == p, "columns of A must match rows of B"
    return [[sum(A[i][k] * B[k][j] for k in range(p)) for j in range(n)] for i in range(m)]


def dot(a, b):
    return sum(x * y for x, y in zip(a, b))


def norm(a):
    return math.sqrt(dot(a, a))


def is_zero(x):
    return abs(x) < EPS


def close(A, B, tol=1e-8):
    return all(abs(a - b) < tol for ra, rb in zip(A, B) for a, b in zip(ra, rb))


# ----------------------------------------------------------------- printing
def fmt(x, digits=3):
    """Format one number: exact for ints and Fractions, rounded for floats."""
    if isinstance(x, Fraction):
        return str(x.numerator) if x.denominator == 1 else f"{x.numerator}/{x.denominator}"
    if isinstance(x, int):
        return str(x)
    if abs(x) < 5 * 10 ** (-digits - 1):
        return "0"
    s = f"{x:.{digits}f}".rstrip("0").rstrip(".")
    return "0" if s in ("-0", "") else s


def show(A, name=None, aug=0, digits=3):
    """Print a matrix (or vector) with brackets. aug = number of augmented columns."""
    rows = A if isinstance(A[0], (list, tuple)) else [[x] for x in A]
    cells = [[fmt(x, digits) for x in row] for row in rows]
    width = max(len(c) for row in cells for c in row)
    n = len(cells[0])
    pad = " " * (len(name) + 3) if name else ""
    for i, row in enumerate(cells):
        parts = []
        for j, c in enumerate(row):
            if aug and j == n - aug:
                parts.append("|")
            parts.append(c.rjust(width))
        lead = f"{name} = " if (name and i == len(cells) // 2) else pad
        print(f"{lead}[ {' '.join(parts)} ]")
    print()


# ----------------------------------------------------------------- lecture 2
def eliminate(A, b=None, verbose=False, partial_pivoting=False):
    """
    Gaussian elimination on a square system. Returns a dict with
        U, c          the upper triangular system U x = c
        pivots        the pivots, in order
        L             the multipliers, stored as a lower triangular matrix
        perm          row order after exchanges (perm[i] = original row now in row i)
        exchanges     how many row exchanges were needed
        singular      True if a pivot position could not be filled
    Rows are exchanged only when a pivot is zero (as in the lecture), unless
    partial_pivoting=True, which always brings up the largest entry (as software does).
    """
    A = copy(A)
    n = len(A)
    b = list(b) if b is not None else None
    L = identity(n)
    perm = list(range(n))
    pivots, exchanges = [], 0

    for k in range(n):
        p = k
        if partial_pivoting:
            p = max(range(k, n), key=lambda r: abs(A[r][k]))
        elif is_zero(A[k][k]):
            below = [r for r in range(k + 1, n) if not is_zero(A[r][k])]
            p = below[0] if below else k
        if is_zero(A[p][k]):
            if verbose:
                print(f"  no pivot available in column {k + 1}: the matrix is singular\n")
            return dict(U=A, c=b, pivots=pivots, L=L, perm=perm, exchanges=exchanges, singular=True)
        if p != k:
            A[k], A[p] = A[p], A[k]
            perm[k], perm[p] = perm[p], perm[k]
            if b is not None:
                b[k], b[p] = b[p], b[k]
            for j in range(k):
                L[k][j], L[p][j] = L[p][j], L[k][j]
            exchanges += 1
            if verbose:
                print(f"  exchange rows {k + 1} and {p + 1}")
        pivots.append(A[k][k])
        for i in range(k + 1, n):
            l = A[i][k] / A[k][k]
            L[i][k] = l
            if is_zero(l):
                continue
            for j in range(k, n):
                A[i][j] = A[i][j] - l * A[k][j]
            if b is not None:
                b[i] = b[i] - l * b[k]
            if verbose:
                print(f"  row {i + 1}  <-  row {i + 1} - ({fmt(l)}) * row {k + 1}")
    return dict(U=A, c=b, pivots=pivots, L=L, perm=perm, exchanges=exchanges, singular=False)


def back_substitute(U, c):
    """Solve U x = c for upper triangular U, from the last equation upward."""
    n = len(U)
    x = [0] * n
    for i in range(n - 1, -1, -1):
        s = c[i] - sum(U[i][j] * x[j] for j in range(i + 1, n))
        x[i] = s / U[i][i]
    return x


def forward_substitute(L, b):
    """Solve L y = b for lower triangular L, from the first equation downward."""
    n = len(L)
    y = [0] * n
    for i in range(n):
        s = b[i] - sum(L[i][j] * y[j] for j in range(i))
        y[i] = s / L[i][i]
    return y


def solve(A, b):
    """Solve A x = b by elimination and back substitution. None if singular."""
    r = eliminate(A, b, partial_pivoting=not isinstance(A[0][0], Fraction))
    return None if r["singular"] else back_substitute(r["U"], r["c"])


def det(A):
    """Determinant = product of the pivots, with a sign flip per row exchange."""
    r = eliminate(A, partial_pivoting=not isinstance(A[0][0], Fraction))
    if r["singular"]:
        return 0
    d = -1 if r["exchanges"] % 2 else 1
    for p in r["pivots"]:
        d = d * p
    return d


# ----------------------------------------------------------------- lecture 3
def inverse(A, verbose=False):
    """Gauss-Jordan: row-reduce [A I] until it reads [I A^-1]. None if singular."""
    n = len(A)
    W = [list(row) + identity(n)[i] for i, row in enumerate(A)]
    for k in range(n):
        p = next((r for r in range(k, n) if not is_zero(W[r][k])), None)
        if p is None:
            return None
        W[k], W[p] = W[p], W[k]
        for i in range(n):
            if i != k and not is_zero(W[i][k]):
                l = W[i][k] / W[k][k]
                W[i] = [a - l * bk for a, bk in zip(W[i], W[k])]
        if verbose:
            show(W, aug=n)
    for k in range(n):
        d = W[k][k]
        W[k] = [a / d for a in W[k]]
    return [row[n:] for row in W]


# ----------------------------------------------------------------- lectures 4-5
def lu(A, partial_pivoting=False):
    """
    Factor PA = LU. Returns (P, L, U, perm). With no row exchanges P is the identity
    and this is the plain A = LU of lecture 4. Returns None if A is singular.
    """
    r = eliminate(A, partial_pivoting=partial_pivoting)
    if r["singular"]:
        return None
    n = len(A)
    P = [[1 if r["perm"][i] == j else 0 for j in range(n)] for i in range(n)]
    return P, r["L"], r["U"], r["perm"]


def lu_solve(factors, b):
    """Solve A x = b from stored factors: L y = P b, then U x = y. Cost about n^2."""
    P, L, U, perm = factors
    pb = [b[p] for p in perm]
    return back_substitute(U, forward_substitute(L, pb))


def ldu(A):
    """A = L D U with ones on both diagonals of L and U. No row exchanges."""
    P, L, U, perm = lu(A)
    n = len(A)
    D = [[U[i][i] if i == j else 0 for j in range(n)] for i in range(n)]
    U1 = [[U[i][j] / U[i][i] for j in range(n)] for i in range(n)]
    return L, D, U1


def permutation_matrix(perm):
    n = len(perm)
    return [[1 if perm[i] == j else 0 for j in range(n)] for i in range(n)]

def rref(A, tol=1e-9):
    """
    Reduced row echelon form of any m by n matrix.
    Returns (R, pivot_columns). Rank = len(pivot_columns).
    """
    A = copy(A)
    m, n = shape(A)
    pivot_cols, r = [], 0
    for c in range(n):
        if r == m:
            break
        p = next((i for i in range(r, m) if abs(A[i][c]) > tol), None)
        if p is None:
            continue                      # no pivot here: column c is a free column
        A[r], A[p] = A[p], A[r]
        piv = A[r][c]
        A[r] = [a / piv for a in A[r]]
        for i in range(m):
            if i != r and abs(A[i][c]) > tol:
                l = A[i][c]
                A[i] = [a - l * b for a, b in zip(A[i], A[r])]
        pivot_cols.append(c)
        r += 1
    A = [[0 if (not isinstance(a, Fraction) and abs(a) < tol) else a for a in row] for row in A]
    return A, pivot_cols


def rank(A):
    return len(rref(A)[1])


def nullspace(A):
    """
    Special solutions of A x = 0: one for each free column.
    Set that free variable to 1, the other free variables to 0, and read the
    pivot variables off the reduced matrix R (they are minus the free column).
    """
    R, pivot_cols = rref(A)
    n = len(A[0])
    free_cols = [j for j in range(n) if j not in pivot_cols]
    basis = []
    for f in free_cols:
        x = [0] * n
        x[f] = 1
        for i, p in enumerate(pivot_cols):
            x[p] = -R[i][f]
        basis.append(x)
    return basis


def in_column_space(A, b):
    """A x = b is solvable exactly when b adds no new direction to the columns."""
    Ab = [list(row) + [bi] for row, bi in zip(A, b)]
    return rank(Ab) == rank(A)


# ----------------------------------------------------------------- robot helpers
def rot2(theta):
    c, s = math.cos(theta), math.sin(theta)
    return [[c, -s], [s, c]]


def hom2(theta, x, y):
    """Planar homogeneous transform: rotate by theta, then translate by (x, y)."""
    c, s = math.cos(theta), math.sin(theta)
    return [[c, -s, x], [s, c, y], [0, 0, 1]]


def arm_points(lengths, angles):
    """Joint positions of a planar arm. angles are relative joint angles in radians."""
    x = y = a = 0.0
    pts = [(x, y)]
    for L, q in zip(lengths, angles):
        a += q
        x, y = x + L * math.cos(a), y + L * math.sin(a)
        pts.append((x, y))
    return pts


def arm_jacobian(lengths, angles):
    """2 by n Jacobian: column i is the hand velocity when only joint i turns at 1 rad/s."""
    pts = arm_points(lengths, angles)
    tx, ty = pts[-1]
    n = len(lengths)
    return [[-(ty - pts[i][1]) for i in range(n)], [tx - pts[i][0] for i in range(n)]]

    print("\n" + title)
    print("-" * len(title))

"""
Lecture 1 - The Geometry of Linear Equations, on a robot.

    python3 code/l01_geometry.py

Covers: row picture, column picture, A x = b, A times x two ways, the 3 by 3
example, and the question "can we solve A x = b for every b?".
Robots: triangulation from two beacons, a two-link arm, a three-thruster vehicle.
"""
import math
from fractions import Fraction
from linalg import (frac, show, fmt, heading, solve, matvec, combine, column, from_columns,
                    transpose, arm_jacobian, det)


def lecture_example():
    heading("1. The lecture's 2 by 2 example")
    A = frac([[2, -1], [-1, 2]])
    b = frac([0, 3])
    show(A, "A")
    show(b, "b")
    x = solve(A, b)
    print("Row picture:    the lines 2x - y = 0 and -x + 2y = 3 cross at", tuple(fmt(v) for v in x))
    c1, c2 = column(A, 0), column(A, 1)
    print("Column picture: x*col1 + y*col2 =", [fmt(v) for v in combine([c1, c2], x)], " which is b\n")


def two_ways_to_multiply():
    heading("2. A times x, two ways")
    A, x = [[2, 5], [1, 3]], [1, 2]
    by_cols = combine([column(A, 0), column(A, 1)], x)      # 1*(2,1) + 2*(5,3)
    by_rows = matvec(A, x)                                   # (2,5).(1,2) and (1,3).(1,2)
    print("by columns:", by_cols, "   by rows:", by_rows, "\n")


def three_by_three():
    heading("3. The lecture's 3 by 3 example")
    A = frac([[2, -1, 0], [-1, 2, -1], [0, -3, 4]])
    for b in ([0, -1, 4], [1, 1, -3]):
        x = solve(A, frac(b))
        print(f"b = {b}  ->  x = {[fmt(v) for v in x]}")
    print("The first b is column 3 itself; the second is column 1 + column 2.\n")


def triangulation():
    heading("4. Robot: triangulation is the row picture")
    robot = (0.8, -1.3)                       # the truth, unknown to the robot
    beacons = [(-3.4, 2.1), (3.3, 1.7)]
    A, b = [], []
    for (p, q) in beacons:
        phi = math.atan2(q - robot[1], p - robot[0])        # the bearing the robot measures
        # the robot lies on the line through the beacon in direction phi:
        #     sin(phi) x - cos(phi) y = sin(phi) p - cos(phi) q
        A.append([math.sin(phi), -math.cos(phi)])
        b.append(math.sin(phi) * p - math.cos(phi) * q)
    show(A, "A")
    show(b, "b")
    print("computed position:", [fmt(v) for v in solve(A, b)], " true position:", robot)

    # the same fix with a 1 degree bearing error, at a wide and at a thin crossing
    eps = math.radians(1.0)
    for name, rb in (("wide crossing", (0.8, -1.3)), ("thin crossing", (0.0, 1.6))):
        true = [math.atan2(q - rb[1], p - rb[0]) for (p, q) in beacons]
        # sign the two errors so they pull the crossing point along the lines (the bad direction)
        measured = [true[0] + eps, true[1] - eps * math.cos(true[0] - true[1])]
        A = [[math.sin(phi), -math.cos(phi)] for phi in measured]
        b = [math.sin(phi) * p - math.cos(phi) * q for phi, (p, q) in zip(measured, beacons)]
        crossing = math.degrees(abs(true[0] - true[1])) % 180
        crossing = min(crossing, 180 - crossing)
        err = math.dist(solve(A, b), rb)
        print(f"{name} ({crossing:.0f} degrees): 1 degree of sensor error becomes {err:.2f} m of position error")
    print()


def arm_columns():
    heading("5. Robot: a two-link arm is the column picture")
    lengths = [2.2, 1.7]
    want = [0.9, 0.45]                         # hand velocity we ask for
    for elbow_deg in (-72, -10, 0):
        q = [math.radians(58), math.radians(elbow_deg)]
        J = arm_jacobian(lengths, q)
        qdot = solve(J, want) if abs(det(J)) > 1e-9 else None
        print(f"elbow at {elbow_deg:4d} deg   columns {[round(v, 2) for v in column(J, 0)]} and "
              f"{[round(v, 2) for v in column(J, 1)]}   joint speeds:",
              "no solution (singular)" if qdot is None else [round(v, 2) for v in qdot])
    print("As the arm straightens the columns line up and the joint speeds blow up.\n")


def thrust_allocation():
    heading("6. Robot: three thrusters, three unknowns")

    def column_of(x, y, ang):                  # one newton at (x, y) pushing along ang
        return [math.cos(ang), math.sin(ang), x * math.sin(ang) - y * math.cos(ang)]

    good = [(-1.3, 0.5, 0.0), (-1.3, -0.5, 0.0), (0.6, -0.7, math.pi / 2)]
    centre = [(-1.3, 0.0, 0.0), (0.0, -0.7, math.pi / 2), (-1.0, -0.6, math.atan2(0.6, 1.0))]
    want = [1.0, 0.0, 1.0]                     # push forward and turn
    for name, layout in (("good layout", good), ("all aimed at the centre", centre)):
        B = from_columns([column_of(*t) for t in layout])
        print(name)
        show(B, "B", digits=2)
        f = solve(B, want) if abs(det(B)) > 1e-9 else None
        print("  thrusts for (Fx, Fy, torque) =", want, ":",
              "no solution, the torque row is all zeros" if f is None else [round(v, 2) for v in f], "\n")


def practice():
    heading("7. Practice problems (answers to P1-P9 on the page)")
    print("P1/P2  x + y = 5, x - y = 1          ->", [fmt(v) for v in solve(frac([[1, 1], [1, -1]]), frac([5, 1]))])
    print("P3     [1 2; 3 4] times (2, -1)      ->", matvec([[1, 2], [3, 4]], [2, -1]),
          " by columns:", combine([[1, 3], [2, 4]], [2, -1]))
    print("P4     k = 4 makes the rows proportional: det [1 2; 2 4] =", det(frac([[1, 2], [2, 4]])))
    A = frac([[2, -1, 0], [-1, 2, -1], [0, -3, 4]])
    for b in ([2, -1, 0], [-1, 1, 1], [4, -1, -4]):
        print(f"P5     b = {b}".ljust(38), "->", [fmt(v) for v in solve(A, frac(b))])
    # P6: beacon A (4, 6) at 45 degrees gives x - y = -2; beacon B (7, 1) at 0 degrees gives y = 1
    robot = solve(frac([[1, -1], [0, 1]]), frac([-2, 1]))
    print("P6     robot position                ->", [fmt(v) for v in robot])
    bearings = [math.degrees(math.atan2(q - float(robot[1]), p - float(robot[0]))) for p, q in ((4, 6), (7, 1))]
    print("       check: bearings from there    ->", [round(v, 6) for v in bearings])
    J = arm_jacobian([1, 1], [0, math.pi / 2])
    print("P7     J =", [[round(v, 6) for v in row] for row in J],
          " joint speeds for (0, 1):", [round(v, 6) for v in solve(J, [0, 1])])
    Jstraight = arm_jacobian([1, 1], [0, 0])
    print("       straight arm columns:", [round(v, 6) for v in column(Jstraight, 0)], [round(v, 6) for v in column(Jstraight, 1)])
    B = frac([[1, 1, 0], [0, 0, 1], [-1, 1, 0]])
    print("P8     thrusts for (2, 1, 1)          ->", [fmt(v) for v in solve(B, frac([2, 1, 1]))], " det B =", fmt(det(B)))
    for rhs in ("2.01", "2.02"):
        x = solve([[Fraction(1), Fraction(1)], [Fraction(1), Fraction("1.01")]], [Fraction(2), Fraction(rhs)])
        print(f"P9     second right-hand side {rhs}   ->", [fmt(v) for v in x])
    print()

Lecture 2 - Elimination with Matrices, on a robot.

    python3 code/l02_elimination.py

Covers: elimination with pivots and multipliers, the augmented matrix and
back-substitution, temporary and permanent failure, reading a product by rows
and by columns, elimination matrices E21 and E32, the associative law and the
order test, permutation matrices on the left and on the right, and a first inverse.
Robots: a two-wheeled robot, and three-wheel omni odometry in four wheel layouts.

import math
from linalg import (frac, show, fmt, heading, eliminate, back_substitute, solve, matmul, matvec, combine,
                    column, identity, inverse, det, close)

A_LECTURE = [[1, 2, 1], [3, 8, 1], [0, 4, 1]]

# ----------------------------------------------------------------- helpers
def elim_matrix(n, i, j, l):
    """E_ij: the identity with -l in row i, column j (1-based). It subtracts l * row j from row i."""
    E = identity(n)
    E[i - 1][j - 1] = -l
    return E


def exchange_matrix(n, i, j):
    """P_ij: the identity with rows i and j exchanged (1-based)."""
    P = identity(n)
    P[i - 1], P[j - 1] = P[j - 1], P[i - 1]
    return P


def augmented(A, b):
    return [list(row) + [bi] for row, bi in zip(A, b)]


def walk(A, b, digits=3):
    """
    Lecture-style elimination on [A | b], printing every step.
    Rows are exchanged only when a pivot position holds zero.
    Returns (U, c, pivots, multipliers, exchanges) or None if elimination fails.
    """
    M = augmented(A, b)
    n = len(A)
    pivots, multipliers, exchanges = [], [], 0
    show(M, "[A|b]", aug=1, digits=digits)
    for k in range(n):
        if abs(M[k][k]) < 1e-10:
            below = [r for r in range(k + 1, n) if abs(M[r][k]) > 1e-10]
            if not below:
                print(f"  zero in pivot position {k + 1} and nothing below to exchange in: PERMANENT FAILURE (singular)")
                print(f"  pivots found: {[fmt(p, digits) for p in pivots]}\n")
                return None
            r = below[0]
            M[k], M[r] = M[r], M[k]
            exchanges += 1
            print(f"  zero in pivot position {k + 1}: exchange rows {k + 1} and {r + 1}")
            show(M, aug=1, digits=digits)
        pivots.append(M[k][k])
        for i in range(k + 1, n):
            l = M[i][k] / M[k][k]
            multipliers.append(l)
            if abs(l) < 1e-10:
                print(f"  step ({i + 1},{k + 1}): entry is already 0, multiplier 0, nothing to do")
                continue
            M[i] = [a - l * p for a, p in zip(M[i], M[k])]
            M[i][k] = M[i][k] * 0
            print(f"  step ({i + 1},{k + 1}): pivot {fmt(M[k][k], digits)}, multiplier {fmt(l, digits)}:  "
                  f"row {i + 1} <- row {i + 1} - ({fmt(l, digits)}) * row {k + 1}")
            show(M, aug=1, digits=digits)
    U = [row[:n] for row in M]
    c = [row[n] for row in M]
    return U, c, pivots, multipliers, exchanges


def back_sub_verbose(U, c, names):
    n = len(U)
    x = [0] * n
    for i in range(n - 1, -1, -1):
        known = sum(U[i][j] * x[j] for j in range(i + 1, n))
        x[i] = (c[i] - known) / U[i][i]
        print(f"  row {i + 1}: {names[i]} = ({fmt(c[i])} - ({fmt(known)})) / {fmt(U[i][i])} = {fmt(x[i])}")
    return x


def r2(v):
    """Round a vector for printing."""
    return [round(float(x), 3) + 0.0 for x in v]

    show(U, "E32 (E21 A)")

R = 0.2                                        # wheel circle radius, metres

def omni_row(alpha):
    """Tangent wheel at angle alpha on a circle of radius R:  s = -sin(a) vx + cos(a) vy + R w."""
    return [-math.sin(alpha), math.cos(alpha), R]


def wheel_row(x, y, beta):
    """Any wheel at (x, y) rolling along the unit direction (cos beta, sin beta)."""
    dx, dy = math.cos(beta), math.sin(beta)
    return [dx, dy, x * dy - y * dx]


def clean(A):
    return [[0.0 if abs(v) < 1e-12 else v for v in row] for row in A]


def omni_derivation_check():
    heading("9. Robot: the omni wheel equation, checked from first principles")
    twist = (0.37, -0.21, 0.9)                 # an arbitrary body velocity (vx, vy, w)
    vx, vy, w = twist
    worst = 0.0
    for deg in (90, 210, 330, 0, 120, 240, 17):
        a = deg * D
        px, py = R * math.cos(a), R * math.sin(a)          # where the wheel touches the floor
        point_velocity = (vx - w * py, vy + w * px)         # velocity of that point of a rigid body
        rolling = (-math.sin(a), math.cos(a))               # the wheel's rolling direction (the tangent)
        direct = rolling[0] * point_velocity[0] + rolling[1] * point_velocity[1]
        formula = sum(c * t for c, t in zip(omni_row(a), twist))
        worst = max(worst, abs(direct - formula))
    print("largest difference between the formula and the direct computation, over 7 wheel angles:", f"{worst:.1e}")
    print("general wheel row agrees with the tangent formula:",
          close([wheel_row(R * math.cos(a), R * math.sin(a), a + math.pi / 2) for a in (0.3, 2.0)],
                [omni_row(a) for a in (0.3, 2.0)]), "\n")

def odometry():
    heading("13. Robot: odometry = solve every tick and add up the motion")
    A = clean([omni_row(d * D) for d in (90, 210, 330)])
    s = [-0.20, 0.25, 0.25]                    # constant readings
    x = y = th = 0.0
    dt, t = 0.01, 0.0
    period = 2 * math.pi / 0.5                 # time for one full turn at 0.5 rad/s
    marks = [period / 4, period / 2, period]
    for _ in range(int(round(period / dt))):
        vx, vy, w = solve(A, s)                # elimination and back-substitution, every tick
        x += (math.cos(th) * vx - math.sin(th) * vy) * dt      # body velocity turned into the world frame
        y += (math.sin(th) * vx + math.cos(th) * vy) * dt
        th += w * dt
        t += dt
        if marks and t >= marks[0] - 1e-9:
            print(f"  t = {t:5.2f} s   x = {x:6.3f} m   y = {y:6.3f} m   heading = {math.degrees(th):6.1f} deg")
            marks.pop(0)
    print("The robot drives a circle of radius 0.6 m and returns (nearly) to its start.\n")

def gauss_jordan(M):
    """
    Reduce [M | I] to [I | M^-1], printing every row step, in the lecture's order:
    clear below the pivots (Gauss), clear above them (Jordan), divide by the pivots.
    Returns (inverse or None, the list of step matrices in the order they were applied).
    """
    n = len(M)
    W = [list(row) + identity(n)[i] for i, row in enumerate(M)]
    steps = []

    def record(E, what):
        steps.append(E)
        print("  " + what)
        show(W, aug=n)

    print("  start with [A | I]")
    show(W, aug=n)
    for k in range(n):                                   # Gauss: downward
        if is_zero(W[k][k]):
            p = next((r for r in range(k + 1, n) if not is_zero(W[r][k])), None)
            if p is None:
                print(f"  no pivot can be found in column {k + 1}: the matrix is singular\n")
                return None, steps
            W[k], W[p] = W[p], W[k]
            E = identity(n)
            E[k], E[p] = E[p], E[k]
            record(E, f"exchange rows {k + 1} and {p + 1}")
        for i in range(k + 1, n):
            l = W[i][k] / W[k][k]
            if is_zero(l):
                continue
            W[i] = [a - l * b for a, b in zip(W[i], W[k])]
            E = identity(n)
            E[i][k] = -l
            record(E, f"row {i + 1}  <-  row {i + 1} - ({fmt(l)}) * row {k + 1}")
    for k in range(n - 1, -1, -1):                       # Jordan: upward
        for i in range(k - 1, -1, -1):
            l = W[i][k] / W[k][k]
            if is_zero(l):
                continue
            W[i] = [a - l * b for a, b in zip(W[i], W[k])]
            E = identity(n)
            E[i][k] = -l
            record(E, f"row {i + 1}  <-  row {i + 1} - ({fmt(l)}) * row {k + 1}")
    for k in range(n):                                   # make every pivot 1
        d = W[k][k]
        if not is_zero(d - 1):
            W[k] = [a / d for a in W[k]]
            E = identity(n)
            E[k][k] = 1 / d
            record(E, f"row {k + 1}  <-  row {k + 1} / ({fmt(d)})")
    return [row[n:] for row in W], steps

    heading("10. Finding the inverse: two systems with the same A")

def row(v, digits=3):
    """One vector on one line."""
    return "(" + ", ".join(fmt(x, digits) for x in v) + ")"


def mat(A):
    """A small matrix on one line, rows separated by semicolons: [1 2; 3 7]."""
    return "[" + "; ".join(" ".join(fmt(x) for x in r) for r in A) + "]"

def two_triangular_passes():
    heading("5. Solving with L and U: forward, then back")
    A = frac([[2, 1], [8, 7]])
    factors = lu(A)
    P, L, U, perm = factors
    for b in ([3, 15], [5, 23]):

def counted_lu(A):
    """Elimination with a counter: one operation per division and per multiply-subtract."""
    A = [list(r) for r in A]
    n, ops = len(A), 0
    for k in range(n - 1):
        for i in range(k + 1, n):
            l = A[i][k] / A[k][k]
            ops += 1                                   # the multiplier
            A[i][k] = l                                # keep it where the zero would go: this is L
            for j in range(k + 1, n):
                A[i][j] -= l * A[k][j]
                ops += 1                               # one multiply-subtract
    return A, ops


def counted_solve(LU, b):
    """Forward then back substitution on the packed factors, with the same counter."""
    n, ops = len(LU), 0
    y = list(b)
    for i in range(n):
        for j in range(i):
            y[i] -= LU[i][j] * y[j]
            ops += 1
    x = y[:]
    for i in range(n - 1, -1, -1):
        for j in range(i + 1, n):
            x[i] -= LU[i][j] * x[j]
            ops += 1
        x[i] /= LU[i][i]
        ops += 1
    return x, ops


def test_matrix(n):
    """A full matrix that needs no row exchanges (large diagonal)."""
    return [[(n + 1.0) if i == j else 1.0 / (1 + abs(i - j)) for j in range(n)] for i in range(n)]


def factor_ops(n):
    return (n ** 3 - n) // 3


def cost_of_elimination():
    heading("6. The cost of elimination")
    print("Strang's estimate: n^2 + (n-1)^2 + ... + 1 is about n^3/3 (the area under x^2).")
    n = 100
    exact_sum = sum(k * k for k in range(1, n + 1))
    print(f"n = {n}: sum of squares = {exact_sum:,}   n^3/3 = {n ** 3 / 3:,.0f}\n")
    print("Counting every division and multiply-subtract in a real run:")
    print("    n    factor A (counted)   (n^3 - n)/3    one right-hand side (counted)    n^2")
    for n in (2, 3, 6, 10, 50, 100):
        A = test_matrix(n)
        b = [float(i + 1) for i in range(n)]
        LU, f = counted_lu(A)
        x, s = counted_solve(LU, b)
        assert max(abs(v - w) for v, w in zip(matvec(A, x), b)) < 1e-8
        print(f"{n:5d} {f:17,d} {factor_ops(n):15,d} {s:22,d} {n * n:16,d}")
    print("So: about n^3/3 to factor A, exactly n^2 for each right-hand side.")
    n, k = 100, 1000
    scratch = k * (factor_ops(n) + n * n)
    once = factor_ops(n) + k * n * n
    print(f"n = {n}, {k} right-hand sides: from scratch {scratch:,}, factor once {once:,}, "
          f"speed-up {scratch / once:.1f} times")
    print(f"With many right-hand sides the speed-up levels off at (n^3/3 + n^2)/n^2, about n/3 + 1 = {n / 3 + 1:.1f}.\n")

def pose_of(T):
    """(x, y, heading in degrees) of a planar transform, with tiny round-off cleaned to zero."""
    tidy = lambda v: 0.0 if abs(v) < 5e-10 else v
    return tidy(T[0][2]), tidy(T[1][2]), tidy(math.degrees(math.atan2(T[1][0], T[0][0])))


def socks_and_shoes_rover():
    heading("8. Robot: undo two moves in reverse order")
    a, turn, d = 3.0, math.radians(70), 2.5
    A = hom2(0.0, a, 0.0)                                   # move A: drive 3 m straight ahead
    B = hom2(turn, d * math.cos(turn), d * math.sin(turn))  # move B: turn 70 degrees, then drive 2.5 m
    AB = matmul(A, B)
    show(A, "A", digits=2)
    show(B, "B", digits=2)
    show(AB, "A B", digits=2)
    print("pose after A then B:      x = %.3f  y = %.3f  heading = %.0f deg" % pose_of(AB))
    right = matmul(AB, matmul(inverse(B), inverse(A)))
    wrong = matmul(AB, matmul(inverse(A), inverse(B)))
    print("undo B first, then A:     x = %.3f  y = %.3f  heading = %.0f deg   (home)" % pose_of(right))
    x, y, th = pose_of(wrong)
    print("undo A first, then B:     x = %.3f  y = %.3f  heading = %.0f deg" % (x, y, th))
    print("miss distance: %.3f m   (formula 2 a sin(turn/2) = %.3f m)" % (math.hypot(x, y), 2 * a * math.sin(turn / 2)))
    straight = hom2(0.0, d, 0.0)
    both = matmul(matmul(A, straight), matmul(inverse(A), inverse(straight)))
    print("with no turn in B the two moves commute and either order works: x = %.3f  y = %.3f\n" % pose_of(both)[:2])

def tridiagonal(m):
    return [[4 if i == j else (1 if abs(i - j) == 1 else 0) for j in range(m)] for i in range(m)]


def banded_factor(m):
    """LU of the 1, 4, 1 matrix keeping only the two diagonals that are not zero."""
    d, l, ops = [4.0], [0.0], 0
    for i in range(1, m):
        l.append(1.0 / d[i - 1])
        d.append(4.0 - l[i])
        ops += 2
    return l, d, ops


def banded_solve(l, d, r):
    m, ops = len(d), 0
    y = [r[0]]
    for i in range(1, m):
        y.append(r[i] - l[i] * y[i - 1])
        ops += 1
    x = [0.0] * m
    x[m - 1] = y[m - 1] / d[m - 1]
    ops += 1
    for i in range(m - 2, -1, -1):
        x[i] = (y[i] - x[i + 1]) / d[i]
        ops += 2
    return x, ops


def spline_rhs(y, h=1.0):
    return [6 * (y[i - 1] - 2 * y[i] + y[i + 1]) / h ** 2 for i in range(1, len(y) - 1)]


def spline_eval(y, M, t):
    """Natural cubic spline with knots at t = 0, 1, 2, ... (h = 1). M includes both end zeros."""
    i = min(int(t), len(y) - 2)
    u = t - i
    a, b = y[i] - M[i] / 6, y[i + 1] - M[i + 1] / 6
    pos = M[i] * (1 - u) ** 3 / 6 + M[i + 1] * u ** 3 / 6 + a * (1 - u) + b * u
    vel = -M[i] * (1 - u) ** 2 / 2 + M[i + 1] * u ** 2 / 2 - a + b
    acc = M[i] * (1 - u) + M[i + 1] * u
    return pos, vel, acc


def spline_path():
    heading("10. Robot: a smooth path through waypoints")
    pts = [(-5.0, -2.4), (-3.3, 0.4), (-1.3, 2.0), (0.5, 0.3), (2.0, -1.9), (3.8, -0.7), (5.2, 2.1)]
    m = len(pts) - 2                                  # unknown accelerations: one per interior waypoint
    T = frac(tridiagonal(m))
    show(T, "T")
    P, L, U, perm = lu(T)
    show(L, "L")
    show(U, "U")
    piv = [U[i][i] for i in range(m)]
    print("pivots:     ", ", ".join(fmt(p) for p in piv), " = ", ", ".join(f"{float(p):.6f}" for p in piv))
    print("multipliers:", ", ".join(fmt(L[i][i - 1]) for i in range(1, m)), " = ",
          ", ".join(f"{float(L[i][i - 1]):.6f}" for i in range(1, m)))
    print(f"limits: pivots -> 2 + sqrt(3) = {2 + math.sqrt(3):.6f}, multipliers -> 2 - sqrt(3) = {2 - math.sqrt(3):.6f}\n")

    xs, ys = [p[0] for p in pts], [p[1] for p in pts]
    factors = lu([[float(v) for v in r] for r in T])      # factored ONCE
    rx, ry = spline_rhs(xs), spline_rhs(ys)
    Mx = [0.0] + lu_solve(factors, rx) + [0.0]            # the same factors serve x ...
    My = [0.0] + lu_solve(factors, ry) + [0.0]            # ... and y
    print("right-hand side for x:", row(rx, 1))
    print("right-hand side for y:", row(ry, 1))
    print("accelerations Mx:", row(Mx, 3))
    print("accelerations My:", row(My, 3))
    print("largest acceleration: %.2f m/s^2" % max(math.hypot(a, b) for a, b in zip(Mx, My)))

    # check the spline: hits every waypoint, velocity and acceleration continuous at every joint
    e = 1e-9
    hit = max(abs(spline_eval(xs, Mx, i)[0] - xs[i]) for i in range(len(xs)))
    jump_v = max(abs(spline_eval(ys, My, i - e)[1] - spline_eval(ys, My, i + e)[1]) for i in range(1, len(ys) - 1))
    jump_a = max(abs(spline_eval(ys, My, i - e)[2] - spline_eval(ys, My, i + e)[2]) for i in range(1, len(ys) - 1))
    print(f"passes through waypoints (largest miss {hit:.1e}); velocity jump at joints {jump_v:.1e}; "
          f"acceleration jump {jump_a:.1e}")
    print("midway between waypoints 2 and 3 the robot is at (%.3f, %.3f)"
          % (spline_eval(xs, Mx, 2.5)[0], spline_eval(ys, My, 2.5)[0]))

    # the banded version gives the same answer for a handful of operations
    l, d, f_ops = banded_factor(m)
    bx, s_ops = banded_solve(l, d, rx)
    print("banded solve agrees with the full one:", close([bx], [Mx[1:-1]]))
    print(f"banded cost for {m} unknowns: factor {f_ops} operations, each solve {s_ops} (a full solve needs n^2 = {m * m})")
    big = 500
    print(f"banded cost for {big} unknowns: factor {2 * (big - 1):,}, each solve {3 * big - 2:,} (full: factor "
          f"{factor_ops(big):,}, each solve {big * big:,})")

    # drag one waypoint: the matrix is untouched, only three right-hand side entries change
    ys2 = ys[:]
    ys2[3] = 1.5
    ry2 = spline_rhs(ys2)
    My2 = [0.0] + lu_solve(factors, ry2) + [0.0]
    print("\nmove waypoint 3 from y = 0.3 to y = 1.5:")
    print("new right-hand side for y:", row(ry2, 1), "  (T, L and U unchanged)")
    print("new accelerations My:     ", row(My2, 3), "\n")


def six_permutations():
    heading("11. Permutations")
    names = {(0, 1, 2): "I", (1, 0, 2): "P12", (2, 1, 0): "P13", (0, 2, 1): "P23", (1, 2, 0): "C", (2, 0, 1): "C2"}
    order = list(names)
    mats = {p: permutation_matrix(list(p)) for p in order}
    key = {tuple(map(tuple, M)): p for p, M in mats.items()}
    for p in order:
        print(f"{names[p]:>3}: rows {mats[p]}")
    print("\nproduct table (row label times column label):")
    print("      " + " ".join(f"{names[q]:>4}" for q in order))
    for p in order:
        cells = [names[key[tuple(map(tuple, matmul(mats[p], mats[q])))]] for q in order]
        print(f"{names[p]:>4}  " + " ".join(f"{c:>4}" for c in cells))
    print("Every product is one of the six.")
    a, b = mats[(1, 0, 2)], mats[(1, 2, 0)]
    print("P12 C =", names[key[tuple(map(tuple, matmul(a, b)))]], "  but  C P12 =",
          names[key[tuple(map(tuple, matmul(b, a)))]], "  (order matters)")
    for p in order:
        assert matmul(transpose(mats[p]), mats[p]) == identity(3)
    print("inverse = transpose for all six:",
          ", ".join(f"{names[p]}^-1 = {names[key[tuple(map(tuple, transpose(mats[p])))]]}" for p in order))
    print("how many: 3 by 3 ->", len(list(permutations(range(3)))), "  4 by 4 ->", len(list(permutations(range(4)))),
          f"  n by n -> n!  (5! = {math.factorial(5)}, 6! = {math.factorial(6)})")

    commands = [10, 20, 30]                         # what the controller sends on outputs 1, 2, 3
    C = mats[(1, 2, 0)]
    got = matvec(C, commands)
    print("\nmotors wired through C receive", got, "instead of", commands)
    fix = matvec(transpose(C), commands)
    print("sending C^T times the commands,", fix, ", repairs it: the motors receive", matvec(C, fix), "\n")

def sig(x, digits=3):
    """Round x to `digits` significant digits, halves away from zero (as a calculator does)."""
    if x == 0:
        return 0.0
    d = Decimal(x)                                   # the exact value of the float
    q = Decimal(1).scaleb(d.adjusted() - digits + 1)
    return float(d.quantize(q, rounding=ROUND_HALF_UP))


def solve_3_digits(a, b, c, d, e, f):
    """Solve [a b; c d] (x, y) = (e, f) by elimination, pivot a, rounding every result to 3 digits."""
    l = sig(c / a)                                   # the multiplier
    d2 = sig(d - sig(l * b))                         # new entry of row 2
    f2 = sig(f - sig(l * e))                         # new right-hand side of row 2
    y = sig(f2 / d2)
    x = sig(sig(e - sig(b * y)) / a)
    return x, y, dict(l=l, d2_exact=d - sig(l * b), d2=d2, f2_exact=f - sig(l * e), f2=f2)

def forward_eliminate(A, b):
    """
    Forward elimination on the augmented matrix [A | b] of any shape.
    Returns the reduced rows [U | c] and the list of pivot positions.
    A row of U that is all zeros reads 0 = c_i: fine if c_i is 0, impossible otherwise.
    """
    M = [list(row) + [bi] for row, bi in zip(A, b)]
    m, n = len(A), len(A[0])
    pivots, r = [], 0
    for c in range(n):
        if r == m:
            break
        p = next((i for i in range(r, m) if abs(M[i][c]) > 1e-9), None)
        if p is None:
            continue
        M[r], M[p] = M[p], M[r]
        for i in range(r + 1, m):
            l = M[i][c] / M[r][c]
            M[i] = [a - l * k for a, k in zip(M[i], M[r])]
        pivots.append((r, c))
        r += 1
    return M, pivots


def one_solution(A, b):
    """One solution of A x = b with every free variable set to 0, or None if there is none."""
    n = len(A[0])
    R, piv = rref([list(row) + [bi] for row, bi in zip(A, b)])
    if n in piv:                      # a pivot in the b column means a row 0 = non-zero
        return None
    x = [0] * n
    for i, p in enumerate(piv):
        x[p] = R[i][n]
    return x


def intersection(S, T):
    """
    Vectors that lie in both span(S) and span(T), S and T being lists of vectors.
    A vector is in both when  S c = T d,  that is  [S  -T] (c, d) = 0:  a nullspace.
    """
    M = from_columns(S + [vscale(t, -1) for t in T])
    return [combine(S, z[:len(S)]) for z in nullspace(M)]

def exact(M):

    return [[Fraction(round(x, 9)).limit_denominator(1000) for x in row] for row in M]


def echelon(A, verbose=False, tol=1e-9):
    """
    Forward elimination on any m by n matrix, exactly as in the lecture.
    Walk across the columns. If a column has a non-zero entry at or below the
    current row, that entry is a pivot: clear everything under it and step down
    one row. If it has none, the column is FREE: stay on the same row and move
    right. Returns (U, pivot_columns).
    """
    A = copy(A)
    m, n = shape(A)
    pivot_cols, r = [], 0
    for c in range(n):
        if r == m:
            break
        p = next((i for i in range(r, m) if abs(A[i][c]) > tol), None)
        if p is None:
            if verbose:
                print(f"  column {c + 1}: zeros from row {r + 1} down, no pivot here -> FREE column, move right")
            continue
        if p != r:
            A[r], A[p] = A[p], A[r]
            if verbose:
                print(f"  column {c + 1}: zero in the pivot position, exchange rows {r + 1} and {p + 1}")
        if verbose:
            print(f"  column {c + 1}: pivot {fmt(A[r][c])} in row {r + 1}")
        for i in range(r + 1, m):
            if abs(A[i][c]) > tol:
                l = A[i][c] / A[r][c]
                A[i] = [a - l * b for a, b in zip(A[i], A[r])]
                if verbose:
                    print(f"      row {i + 1} <- row {i + 1} - ({fmt(l)}) * row {r + 1}   =  {vec(A[i])}")
        pivot_cols.append(c)
        r += 1
    A = [[0 if (not isinstance(a, Fraction) and abs(a) < tol) else a for a in row] for row in A]
    return A, pivot_cols


def solve_from_free(U, pivot_cols, free_values):
    """
    Back-substitution for U x = 0. free_values maps a free column to the value
    chosen for that variable. The pivot variables are then forced, last one first.
    """
    n = len(U[0])
    x = [0] * n
    for col, value in free_values.items():
        x[col] = value
    for row in range(len(pivot_cols) - 1, -1, -1):
        pc = pivot_cols[row]
        rest = sum(U[row][j] * x[j] for j in range(pc + 1, n))
        x[pc] = -rest / U[row][pc]
    return x


def blocks(R, pivot_cols):
    """The F block of R (free columns, pivot rows) and the nullspace matrix N."""
    n = len(R[0])
    free_cols = [j for j in range(n) if j not in pivot_cols]
    F = [[R[i][f] for f in free_cols] for i in range(len(pivot_cols))]
    N = from_columns(nullspace(R)) if free_cols else None
    return F, N, free_cols


def thruster_column(x, y, ang):
    """One newton from a thruster at (x, y) metres pushing along ang: [Fx, Fy, torque]."""
    return [math.cos(ang), math.sin(ang), x * math.sin(ang) - y * math.cos(ang)]


def particular(B, b):
    """One solution of B f = b: every free variable set to 0, pivot variables read off rref([B b])."""
    n = len(B[0])
    R, piv = rref([list(row) + [bi] for row, bi in zip(B, b)])
    if n in piv:
        return None                      # a pivot in the last column: 0 = non-zero, no solution
    f = [0] * n
    for i, p in enumerate(piv):
        f[p] = R[i][n]
    return f

def arm_report(name, degrees, exact_numbers=False):
    q = [math.radians(d) for d in degrees]
    J = arm_jacobian(ARM_LINKS, q)
    if exact_numbers:
        J = exact(J)
    tip = arm_points(ARM_LINKS, q)[-1]
    R, piv = rref(J)
    free = [j for j in range(4) if j not in piv]
    print(f"{name}: joint angles {degrees} degrees, hand at ({tip[0]:.2f}, {tip[1]:.2f}) m")
    show(J, "J", digits=2)
    show(R, "R", digits=2)
    print(f"  rank = {len(piv)}   pivot joints: {names(piv)}   free joints: {names(free)}   redundancy n - r = {4 - len(piv)}")
    for k, s in enumerate(nullspace(J)):
        print(f"  special solution s{k + 1} = {vec(s, 2)} rad/s    J s{k + 1} = {vec(matvec(J, s), 2)} m/s")
    print()
    return J, R, piv, free


def four_joint_arm():
    heading("9. Robot: a four-joint planar arm (2 by 4 Jacobian)")
    print("Column i of J = hand velocity (m/s) when only joint i turns at 1 rad/s")
    print("            = the arrow from joint i to the hand, turned a quarter turn: (-(yh - yi), xh - xi).")
    print("Two rows (hand vx, vy), four columns (four joints).\n")
    J, R, piv, free = arm_report("Staircase pose", STAIRCASE, exact_numbers=True)

    F, N, _ = blocks(R, piv)
    print("x_pivot = -F x_free as a control rule. Choose the free joint speeds, read off the pivot ones:")
    qfree = [Fraction(2, 5), Fraction(-1, 5)]        # joints 3 and 4, rad/s
    qpiv = [-sum(F[i][j] * qfree[j] for j in range(2)) for i in range(2)]
    qdot = [qpiv[0], qpiv[1], qfree[0], qfree[1]]
    print(f"  free joints 3, 4 at {vec([float(v) for v in qfree])} rad/s  ->  pivot joints 1, 2 at {vec([float(v) for v in qpiv])} rad/s")
    print(f"  q_dot = {vec([float(v) for v in qdot])}   hand velocity J q_dot = {vec(matvec(J, qdot))} m/s\n")

    print("A self-motion that lasts: follow q_dot = 0.5 * s1(q), recomputing s1 as the arm moves (1 s, 1 ms steps).")
    for correct in (False, True):
        q = [math.radians(d) for d in STAIRCASE]
        target = arm_points(ARM_LINKS, q)[-1]
        dt = 0.001
        for _ in range(1000):
            s1 = nullspace(arm_jacobian(ARM_LINKS, q))[0]
            q = [qi + 0.5 * si * dt for qi, si in zip(q, s1)]
            if correct:
                # pull the hand back onto the target: solve (J J^T) lam = error, then q += J^T lam
                Jn = arm_jacobian(ARM_LINKS, q)
                tip = arm_points(ARM_LINKS, q)[-1]
                e = [target[0] - tip[0], target[1] - tip[1]]
                a, b_, d = (sum(x * x for x in Jn[0]), sum(x * y for x, y in zip(Jn[0], Jn[1])), sum(y * y for y in Jn[1]))
                det2 = a * d - b_ * b_
                lam = [(d * e[0] - b_ * e[1]) / det2, (a * e[1] - b_ * e[0]) / det2]
                q = [qi + Jn[0][i] * lam[0] + Jn[1][i] * lam[1] for i, qi in enumerate(q)]
        tip = arm_points(ARM_LINKS, q)[-1]
        drift = math.hypot(tip[0] - target[0], tip[1] - target[1])
        label = "with the correction step   " if correct else "plain stepping, no correction"
        print(f"  {label}: joints now at {[round(math.degrees(v), 1) for v in q]} degrees, hand moved {drift * 1000:.3f} mm")
    print("  The joints turned by tens of degrees and the hand stayed put.\n")

    arm_report("Reaching pose", REACHING)
    arm_report("Straight arm", STRAIGHT)
    print("Straight arm: all four columns are parallel, the rank drops to 1 (a singularity),")
    print("and a third special solution appears. The hand can now only move sideways.\n")

def four_thrusters():
    heading("10. Robot: a four-thruster vehicle (3 by 4 thruster matrix)")
    print("Column i of B = [Fx, Fy, torque] from 1 N of thruster i = [cos a, sin a, x sin a - y cos a].")
    print("Rows: forward force (N), side force (N), turning torque (N m). Columns: four thrusters.\n")

    B = exact(from_columns([thruster_column(*t) for t in VEHICLE["two pairs"]]))
    print("Layout 'two pairs': T1 port propeller, T2 bow thruster, T3 starboard propeller, T4 stern thruster")
    show(B, "B")
    U, piv = echelon(B, verbose=True)
    print()
    R, piv = rref(B)
    show(R, "R")
    s = nullspace(B)[0]
    print(f"rank = {len(piv)}   pivot thrusters: {names(piv)}   free thruster: 4   n - r = {4 - len(piv)}")
    print(f"special solution s = {vec(s)} N    B s = {vec(matvec(B, s))}")
    print("  T1 and T3 at -2 and +2 N twist the hull by", fmt(B[2][0] * s[0] + B[2][2] * s[2]), "N m;")
    print("  T2 and T4 at -1 and +1 N twist it by", fmt(B[2][1] * s[1] + B[2][3] * s[3]), "N m. Net force and torque: zero.\n")

    for want in ([1, 0, 0], [0, 1, 0]):
        b = frac(want)
        p = particular(B, b)
        print(f"wanted (Fx, Fy, torque) = {want}")
        print(f"  free thruster set to 0:  f = {vec(p)}   total thrust {fmt(sum(abs(v) for v in p))} N")
        best = None
        for c in (Fraction(1, 4), Fraction(1, 2), Fraction(1)):
            f = [pi + c * si for pi, si in zip(p, s)]
            effort = sum(abs(v) for v in f)
            print(f"  add {fmt(c):>3} * s:            f = {vec(f)}   total thrust {fmt(effort)} N   B f = {vec(matvec(B, f))}")
        # least total thrust: sum |p + c s| is piecewise linear in c, so the best c makes one thruster exactly zero
        cands = [-pi / si for pi, si in zip(p, s) if si != 0]
        best = min(cands, key=lambda c: (sum(abs(pi + c * si) for pi, si in zip(p, s)), abs(c)))
        f = [pi + best * si for pi, si in zip(p, s)]
        print(f"  least total thrust at c = {fmt(best)}:  f = {vec(f)}   total thrust {fmt(sum(abs(v) for v in f))} N\n")

    print("Losing thrusters (delete columns and count the pivots again):")
    for name, layout in VEHICLE.items():
        B = exact(from_columns([thruster_column(*t) for t in layout]))
        s = nullspace(B)[0]
        print(f"  layout '{name}': special solution {vec(s)}")
        for dead in range(4):
            keep = [j for j in range(4) if j != dead]
            r = rank([[row[j] for j in keep] for row in B])
            verdict = "full control survives" if r == 3 else "a direction is lost"
            print(f"     T{dead + 1} fails: rank {r}  ({verdict};  entry {dead + 1} of s is {fmt(s[dead])})")
    B = exact(from_columns([thruster_column(*t) for t in VEHICLE["two pairs"]]))
    for dead in ((0, 2), (1, 3), (0, 1)):
        keep = [j for j in range(4) if j not in dead]
        r = rank([[row[j] for j in keep] for row in B])
        print(f"  'two pairs' with T{dead[0] + 1} and T{dead[1] + 1} both failed: rank {r}")
    print("Rule: a thruster can be lost without losing control exactly when its entry in s is not zero.")
    keep = [0, 1, 2]
    p = particular([[row[j] for j in keep] for row in B], frac([0, 1, 0]))
    print(f"After T4 fails, a 1 N side push needs f = {vec(p)}: total thrust {fmt(sum(abs(v) for v in p))} N, and no choice left.\n")

    second = exact(from_columns([thruster_column(*t) for t in VEHICLE["three astern, one at the bow"]]))
    print("Layout 'three astern, one at the bow':")
    show(second, "B")
    R2, piv2 = rref(second)
    show(R2, "R")
    print(f"pivot columns {names(piv2)}, free column {names([j for j in range(4) if j not in piv2])}: the free column is not always the last one.\n")

    print("Rank is not the number of motors. Four thrusters on the centre line, all pointing forward:")
    same = exact(from_columns([thruster_column(x, 0.0, 0.0) for x in (-1.3, -0.5, 0.3, 1.0)]))
    show(same, "B")
    print(f"rank = {rank(same)}: four motors, one direction of control, n - r = {4 - rank(same)} ways to waste fuel.\n")


def repeated_sensor():
    heading("11. A zero row is a repeated equation: the sensor that adds nothing")
    W = 0.4                                         # distance between the wheels, m
    H = frac([[1, -W / 2], [1, W / 2], [1, 0]])
    print("A two-wheeled robot moves with forward speed v and turn rate w. Three sensors:")
    print("  left wheel  = v - (W/2) w      right wheel = v + (W/2) w      ground camera = v      (W = 0.4 m)")
    show(H, "H")
    U, piv = echelon(H, verbose=True)
    print()
    show(U, "U")
    print(f"rank {len(piv)}: the camera row became a zero row because it is (left + right) / 2.")
    print("It adds no new information about (v, w). The same row operations on the readings turn it into a check:")
    for left, right, cam in ((0.9, 1.1, 1.0), (0.9, 1.4, 1.0)):
        residual = cam - 0.5 * (left + right)
        note = "consistent" if abs(residual) < 1e-9 else "not zero: a wheel is slipping or a sensor is wrong"
        print(f"  readings (left, right, camera) = ({left}, {right}, {cam})  ->  0 = {fmt(residual)}   {note}")
    print()

11. A zero row is a repeated equation: the sensor that adds nothing
-------------------------------------------------------------------
A two-wheeled robot moves with forward speed v and turn rate w. Three sensors:
  left wheel  = v - (W/2) w      right wheel = v + (W/2) w      ground camera = v      (W = 0.4 m)
    [    1 -1/5 ]
H = [    1  1/5 ]
    [    1    0 ]

  column 1: pivot 1 in row 1
      row 2 <- row 2 - (1) * row 1   =  (0, 2/5)
      row 3 <- row 3 - (1) * row 1   =  (0, 1/5)
  column 2: pivot 2/5 in row 2
      row 3 <- row 3 - (1/2) * row 2   =  (0, 0)

    [    1 -1/5 ]
U = [    0  2/5 ]
    [    0    0 ]

rank 2: the camera row became a zero row because it is (left + right) / 2.
